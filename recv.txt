.PHONY: all client server clean

USER := $(shell whoami)

all: client server certs

build: common client certs

user_setup:
	@sudo groupadd nfspadmin
	@sudo usermod -aG nfspadmin $(USER)

dir-setup: user_setup
	@sudo mkdir -p /etc/nfsp
	@sudo mkdir -p /etc/nfsp/certs
	@sudo chgrp -R nfspadmin /etc/nfsp
	@sudo find /etc/nfsp -type d -exec chmod 775 {} \;
	@sudo find /etc/nfsp -type f -exec chmod 664 {} \;
	@sudo chown -R $(USER):nfspadmin /etc/nfsp

install: dir-setup all
	@echo "Installing nfsp_client and nfsp_server to /usr/bin..."
	sudo install -Dm755 build/client/nfsp_client /usr/bin/nfsp_client
	sudo install -Dm755 build/server/nfsp_server /usr/bin/nfsp_server

	@echo "Installing certificates to /etc/nfsp/certs..."
	sudo install -m644 certs/*.pem /etc/nfsp/certs/

	@echo "Fixing permissions and group ownership..."
	sudo chgrp -R nfspadmin /etc/nfsp
	sudo chmod -R g+rw,o+r /etc/nfsp
	sudo find /etc/nfsp -type d -exec chmod g+s {} \;


client:
	$(MAKE) -C src/client

server:
	$(MAKE) -C src/server

certs:
	mkdir -p certs
	# Server certificate
	openssl genpkey -algorithm ED25519 -out certs/server_key.pem
	openssl req -new -x509 -key certs/server_key.pem -out certs/server_cert.pem -days 365 \
		-subj "/CN=NFSPServer"

	# Client certificate
	openssl genpkey -algorithm ED25519 -out certs/client_key.pem
	openssl req -new -x509 -key certs/client_key.pem -out certs/client_cert.pem -days 365 \
		-subj "/CN=NFSPClient"


clean:
	$(MAKE) -C src/client clean
	$(MAKE) -C src/server clean
